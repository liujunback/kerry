安装python

安装网络下载工具
		yum -y install wget
		
下载python包（3.6.5）
		wget https://www.python.org/ftp/python/3.6.5/Python-3.6.5.tar.xz
		
安装python的依赖包gcc
		yum install -y gcc patch libffi-devel python-devel  zlib-devel bzip2-devel openssl-devel ncurses-devel sqlite-devel readline-devel tk-devel gdbm-devel db4-devel libpcap-devel xz-devel

解压python包
		tar -xvf Python-3.6.5.tar.xz
		cd Python-3.6.5

创建一个安装目录
	mkdir /usr/local/python3
	
生成 Makefile 文件，用来编译
		./configure --prefix=/usr/local/python3

安装python
		make && make install
		
创建一个软链接python和pip
		ln -s /usr/local/python3/bin/python3 /usr/bin/python3

		ln -s /usr/local/python3/bin/pip3 /usr/bin/pip3
		
		ln -snf /usr/local/python3/bin/python3 /usr/bin/python  修改linux的软连接指向
		
	
		
安装docker

删除旧版本docker
		sudo yum remove docker \
						  docker-client \
						  docker-client-latest \
						  docker-common \
						  docker-latest \
						  docker-latest-logrotate \
						  docker-logrotate \
						  docker-engine
						  
安装docker的安装依赖
		sudo yum install -y yum-utils \
		  device-mapper-persistent-data \
		  lvm2
		  
更新源镜像（阿里云）
		sudo yum-config-manager \
			--add-repo \
			http://mirrors.aliyun.com/docker-ce/linux/centos/docker-ce.repo
			
安装docker
		sudo yum install docker-ce docker-ce-cli containerd.io
	



使用Django框架
	创建一个Django项目		
		python -m django startproject HelloWorld
	
	下载安装iptables-services  
		yum install iptables-services
		
	关闭防火墙
		systemctl stop firewal
	
	运行Django在局域网内
		python3 manage.py runserver 127.0.0.1:8008
	运行内网穿透
		./ding -config=./ding.cfg -subdomain=testtmss 8008
	ngrok内网穿透
	    ./ngrok http 8008
